set(_repository https://github.com/openDAQ/OpcUaModules.git)

function(expect WHAT GOT WANT)
    if (GOT STREQUAL WANT)
        message(STATUS "ok    ${WHAT}: ${GOT}")
    else()
        message(SEND_ERROR "FAIL  ${WHAT}: ${GOT}, not ${WANT}")
    endif()
endfunction()

# resolve <name> <revision> [<triplet>] → <name>_c_*
function(resolve NAME REVISION)
    set(_triplet "${FETCHLAYER_TRIPLET}")
    if (ARGC GREATER 2)
        set(_triplet "${ARGV2}")
    endif()
    set(r_NAME "${NAME}")
    set(r_REVISION "${REVISION}")
    set(r_DEPENDS_REVISIONS "opendaq=3039ee6d068c6c5e6e4d3de367c91820088ab30e")
    set(r_TRIPLET "${_triplet}")
    __FetchLayer_git_resolve("${NAME}" "${_triplet}" r c)
    foreach(_field ORIGIN REF REVISION DEPENDS_REVISIONS TRIPLET IDENTITY_MATCH)
        set(${NAME}_c_${_field} "${c_${_field}}" PARENT_SCOPE)
    endforeach()
endfunction()

# resolve: the commit, exact, the dependencies as requested
FetchLayer_Declare(opcua-sha REVISION ad11c59ad43348f7099b794d8d8c5c94b8e9ccf5 GIT_REPOSITORY ${_repository})
FetchLayer_DeclareStaging(opcua-sha LAYER opcua-sha)
resolve(opcua-sha ad11c59ad43348f7099b794d8d8c5c94b8e9ccf5)
expect("resolve" "${opcua-sha_c_REVISION}" "ad11c59ad43348f7099b794d8d8c5c94b8e9ccf5")
expect("resolve, exact" "${opcua-sha_c_IDENTITY_MATCH}" "EXACT")
expect("resolve, depends as requested" "${opcua-sha_c_DEPENDS_REVISIONS}" "opendaq=3039ee6d068c6c5e6e4d3de367c91820088ab30e")

FetchLayer_Declare(opcua-tag REVISION v4.0.0 GIT_REPOSITORY ${_repository})

expect("commit, written over the declared tag" "${_declared}" "3e02ec1330224b819bfc5342d9b00a303dd4d9ee")
FetchLayer_Declare(opcua-light REVISION v4.0.1 GIT_REPOSITORY ${_repository})
